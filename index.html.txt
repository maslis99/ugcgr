const downloadHighResImage = (imageUrl, filename, targetRatioStr = '9:16') => {
  const img = new Image();
  img.crossOrigin = "anonymous"; 
  img.src = imageUrl;
  
  img.onload = () => {
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    
    let targetW, targetH;
    switch(targetRatioStr) {
        case '1:1': targetW = 1080; targetH = 1080; break;
        case '4:5': targetW = 1080; targetH = 1350; break;
        case '16:9': targetW = 1920; targetH = 1080; break;
        case '9:16': default: targetW = 1080; targetH = 1920; break;
    }

    canvas.width = targetW;
    canvas.height = targetH;

    ctx.imageSmoothingEnabled = true;
    ctx.imageSmoothingQuality = 'high';

    const scale = Math.max(canvas.width / img.naturalWidth, canvas.height / img.naturalHeight);
    
    const w = img.naturalWidth * scale;
    const h = img.naturalHeight * scale;

    const x = (canvas.width - w) / 2;
    const y = (canvas.height - h) / 2;

    ctx.drawImage(img, x, y, w, h);
    
    const link = document.createElement('a');
    link.download = filename;
    link.href = canvas.toDataURL('image/png', 1.0); 
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };
};

const secureCopy = (text, onSuccess) => {
  if (!text) return;
  try {
    const textArea = document.createElement("textarea");
    textArea.value = text;
    textArea.style.position = "fixed";
    textArea.style.left = "-9999px";
    textArea.style.top = "0";
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    const successful = document.execCommand('copy');
    document.body.removeChild(textArea);
    if (successful && onSuccess) onSuccess();
  } catch (err) { console.error('Copy failed', err); }
};

const compressImage = (file, maxWidth = 1024, quality = 0.8) => new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = (event) => {
        const img = new Image();
        img.src = event.target.result;
        img.onload = () => {
            const canvas = document.createElement('canvas');
            let width = img.width;
            let height = img.height;
            if (width > height) { if (width > maxWidth) { height *= maxWidth / width; width = maxWidth; } } 
            else { if (height > maxWidth) { width *= maxWidth / height; height = maxWidth; } }
            canvas.width = width;
            canvas.height = height;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, width, height);
            resolve(canvas.toDataURL('image/jpeg', quality));
        };
        img.onerror = (error) => reject(error);
    };
    reader.onerror = (error) => reject(error);
});

const fileToBase64 = async (file) => {
    if (!file) return null;
    if (file.type.match(/image.*/)) { try { return await compressImage(file); } catch (e) { console.error(e); } }
    return new Promise((resolve, reject) => {
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = () => resolve(reader.result);
        reader.onerror = (error) => reject(error);
    });
};

const urlToBase64 = async (url) => {
    try {
        const response = await fetch(url);
        const blob = await response.blob();
        return new Promise((resolve) => {
            const reader = new FileReader();
            reader.onloadend = () => resolve(reader.result);
            reader.readAsDataURL(blob);
        });
    } catch { return null; }
};

const wait = (ms) => new Promise(resolve => setTimeout(resolve, ms));

const base64ToArrayBuffer = (base64) => {
    const binaryString = window.atob(base64);
    const len = binaryString.length;
    const bytes = new Uint8Array(len);
    for (let i = 0; i < len; i++) bytes[i] = binaryString.charCodeAt(i);
    return bytes.buffer;
};

const pcmToWav = (pcmData, sampleRate = 24000) => {
    const pcm16 = new Int16Array(pcmData);
    const buffer = new ArrayBuffer(44 + pcm16.length * 2);
    const view = new DataView(buffer);
    const writeString = (offset, string) => { for (let i = 0; i < string.length; i++) view.setUint8(offset + i, string.charCodeAt(i)); };
    writeString(0, 'RIFF'); view.setUint32(4, 36 + pcm16.length * 2, true); writeString(8, 'WAVE'); writeString(12, 'fmt ');
    view.setUint32(16, 16, true); view.setUint16(20, 1, true); view.setUint16(22, 1, true); view.setUint32(24, sampleRate, true);
    view.setUint32(28, sampleRate * 2, true); view.setUint16(32, 2, true); view.setUint16(34, 16, true); writeString(36, 'data');
    view.setUint32(40, pcm16.length * 2, true);
    for (let i = 0; i < pcm16.length; i++) view.setInt16(44 + i * 2, pcm16[i], true);
    return new Blob([view], { type: 'audio/wav' });
};

// --- API FUNCTIONS ---
const callApi = async (payload, isText = false) => {
    const targetModel = isText ? "gemini-2.5-flash-preview-09-2025" : "gemini-2.5-flash-image-preview";
    const url = `https://generativelanguage.googleapis.com/v1beta/models/${targetModel}:generateContent?key=${apiKey}`;
    
    for(let attempt = 0; attempt < 3; attempt++) {
        try {
            const res = await fetch(url, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(payload) });
            if (!res.ok) {
                if (res.status === 429 || res.status >= 500) {
                    await wait(500 * (attempt + 1));
                    continue; 
                }
                throw new Error(`API Error: ${res.status}`);
            }
            const data = await res.json();
            if (!data.candidates || !data.candidates[0] || !data.candidates[0].content) {
                 await wait(500);
                 continue;
            }
            if(isText) return data.candidates[0].content.parts[0].text;
            const imgData = data.candidates[0].content.parts.find(p => p.inlineData);
            if (imgData) return `data:image/png;base64,${imgData.inlineData.data}`;
        } catch(e) {
            if (attempt === 2) return null;
            await wait(500);
        }
    }
    return null;
};

const callTtsApi = async (payload) => {
    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;
    const res = await fetch(url, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(payload) });
    if (!res.ok) throw new Error(`API Error: ${res.status}`);
    const data = await res.json();
    if (!data.candidates?.[0]) throw new Error("Gagal membuat suara.");
    const part = data.candidates[0].content.parts[0];
    return { audioData: part.inlineData.data, sampleRate: 24000 }; 
};

const generateImageToImageAPI = async (prompt, base64Image, additionalImages = []) => {
  const parts = [{ text: prompt }, { inlineData: { mimeType: 'image/png', data: base64Image.split(',')[1] } }];
  additionalImages.forEach(img => { 
      if (img.data) parts.push({ inlineData: { mimeType: 'image/png', data: img.data.split(',')[1] } }); 
  });
  try {
    const payload = { contents: [{ parts }], generationConfig: { responseModalities: ["IMAGE"] } };
    return await callApi(payload, false);
  } catch (e) { return null; }
};

const generateTextAPI = async (prompt) => {
    const payload = { contents: [{ parts: [{ text: prompt }] }] };
    return await callApi(payload, true);
};

      if(voState.isPlaying) { audioRef.current.pause(); setVoState(p=>({...p, isPlaying:false})); }
      else { audioRef.current.play(); setVoState(p=>({...p, isPlaying:true})); audioRef.current.onended = () => setVoState(p=>({...p, isPlaying:false})); }
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-300 font-sans pb-20 selection:bg-yellow-500/30">
      
      {/* Custom Alert Modal Replacer */}
      {alertMessage && (
        <div className="fixed inset-0 z-[100] bg-black/80 backdrop-blur-sm flex items-center justify-center p-4">
            <div className="bg-slate-900 border border-yellow-500 w-full max-w-sm rounded-xl p-6 shadow-2xl text-center animate-in zoom-in-95 duration-200">
                <AlertCircle size={48} className="text-red-500 mx-auto mb-4" />
                <h3 className="text-lg font-bold text-white mb-6">{alertMessage}</h3>
                <button onClick={() => setAlertMessage(null)} className="bg-yellow-600 hover:bg-yellow-500 text-white py-2 px-8 rounded-full font-bold transition-colors w-full">
                    Tutup
                </button>
            </div>
        </div>
      )}

      <nav className="bg-slate-900 border-b border-slate-800 sticky top-0 z-50 shadow-lg">
          <div className="max-w-7xl mx-auto px-4 h-16 flex items-center gap-3">
              <div className="bg-yellow-600 p-2 rounded-lg text-white"><Zap size={20} fill="currentColor"/></div>
              <div>
                  <h1 className="text-white font-bold text-lg tracking-tight">GENERATOR TIM <span className="text-yellow-500">GREEN SAFIR</span></h1>
                  <p className="text-[10px] text-slate-500 font-medium">Tools Sulap AFFILIATE</p>
              </div>
          </div>
      </nav>

      <main className="max-w-7xl mx-auto px-4 py-8 grid grid-cols-1 md:grid-cols-12 gap-8">
        <div className="md:col-span-5 space-y-2">
            <StepPanel number="1" title="Upload Produk" icon={Upload}>
                <div className="relative aspect-square bg-slate-950 rounded-lg border-2 border-dashed border-slate-800 hover:border-yellow-500/50 transition-all flex flex-col items-center justify-center group cursor-pointer overflow-hidden">
                    {mainPhoto ? (
                        <>
                            <img src={mainPhoto} className="w-full h-full object-contain p-4"/>
                            <button onClick={(e)=>{e.stopPropagation(); setMainPhoto(null); setMainPhotoFile(null);}} className="absolute top-2 right-2 bg-red-500/80 p-1.5 rounded text-white hover:bg-red-500 z-20"><X size={14}/></button>
                        </>
                    ) : (
                        <>
                            <Upload size={32} className="text-slate-600 mb-2 group-hover:text-yellow-500 transition-colors"/>
                            <p className="text-xs text-slate-500 font-medium">Klik Upload Foto Utama</p>
                        </>
                    )}
                    <input type="file" accept="image/*" className="absolute inset-0 opacity-0 cursor-pointer" onChange={e => handlePhotoUpload(e, 'main')} disabled={!!mainPhoto}/>
                </div>
                <div className="space-y-2">
                    <p className="text-[10px] font-bold text-slate-500 uppercase">Foto Pendukung (Opsional)</p>
                    {additionalPhotos.map(p => (
                        <div key={p.id} className="flex gap-2 items-center bg-slate-950 p-2 rounded border border-slate-800">
                            <img src={p.url} className="w-8 h-8 rounded object-cover"/>
                            <input placeholder="Keterangan (misal: Background)" className="flex-1 bg-transparent text-xs outline-none" onChange={e => { const newPhotos = additionalPhotos.map(ap => ap.id === p.id ? {...ap, desc: e.target.value} : ap); setAdditionalPhotos(newPhotos); }}/>
                            <button onClick={() => handleRemoveAdditional(p.id)} className="text-slate-500 hover:text-red-400"><Trash2 size={14}/></button>
                        </div>
                    ))}
                    {additionalPhotos.length < 2 && (
                        <label className="flex items-center justify-center gap-2 py-2 border border-slate-800 rounded text-xs text-slate-500 hover:bg-slate-800 cursor-pointer"><Plus size={12}/> Tambah Foto Lain<input type="file" className="hidden" onChange={e => { const file = e.target.files[0]; if(file) setAdditionalPhotos([...additionalPhotos, {id: Date.now(), url: URL.createObjectURL(file), file: file, desc: ''}]); }}/></label>
                    )}
                </div>
            </StepPanel>

            <StepPanel number="2" title="Jelaskan Produk Anda & Konsep Foto" icon={FileText}>
                <div className="animate-in fade-in">
                    <div className="flex justify-end mb-2">
                        <button onClick={handleApplyRecommendation} disabled={isGeneratingDesc} className="text-[10px] text-yellow-400 flex items-center gap-1 hover:text-white">{isGeneratingDesc ? <RefreshCw size={10} className="animate-spin"/> : <Sparkles size={10}/>} {isGeneratingDesc ? 'Menganalisis...' : 'Auto-Generate'}</button>
                    </div>
                    <textarea value={productInfo} onChange={e => setProductInfo(e.target.value)} className="w-full bg-slate-950 border border-slate-800 rounded-lg p-3 text-sm focus:border-yellow-500 outline-none h-24 resize-none" placeholder="Jelaskan produkmu..."/>
                    <div className="mt-3">
                        <label className="text-[10px] font-bold text-slate-500 uppercase mb-1 block">Rasio Gambar</label>
                        <select value={ratio} onChange={e=>setRatio(e.target.value)} className="w-full bg-slate-950 border border-slate-800 rounded p-2 text-xs outline-none">
                            <option value="9:16">9:16 (Story/Tiktok)</option><option value="4:5">4:5 (Feed Portrait)</option><option value="1:1">1:1 (Feed/Square)</option><option value="16:9">16:9 (Landscape/Youtube)</option>
                        </select>
                    </div>
                </div>
            </StepPanel>

            <StepPanel number="3" title="Model AI" icon={User}>
                {models.length < 2 && (
                    <label className="group block w-full text-center py-4 border-2 border-dashed border-slate-700 rounded-xl text-xs text-slate-400 cursor-pointer hover:border-yellow-500 hover:bg-slate-900/50 hover:text-yellow-400 transition-all mb-4 relative overflow-hidden">
                        <div className="flex flex-col items-center justify-center gap-1 relative z-10"><Plus size={16} className="mb-1 text-slate-500 group-hover:text-yellow-500 transition-colors"/><span className="font-bold">UPLOAD WAJAH MODEL SENDIRI</span></div>
                        <input type="file" className="hidden" onChange={e => handlePhotoUpload(e, 'model')}/>
                    </label>
                )}
                {models.length > 0 ? (
                    <div className="animate-in fade-in slide-in-from-bottom-2 duration-300">
                        <div className="grid grid-cols-2 gap-2 mb-4">
                            {models.map(m => (<div key={m.id} className="relative aspect-square bg-slate-950 rounded border border-slate-800 group"><img src={m.url} className="w-full h-full object-cover"/><button onClick={() => setModels(models.filter(x => x.id !== m.id))} className="absolute top-1 right-1 bg-black/60 hover:bg-red-600 text-white p-1 rounded-full"><X size={10}/></button></div>))}
                        </div>
                        <div className="space-y-4 border-t border-slate-800 pt-4">
                            {models.map((m, index) => (
                                <div key={m.id} className="flex gap-3 mb-3 p-2 border border-slate-800 rounded bg-slate-900/50">
                                    <img src={m.url} className="w-12 h-12 object-cover rounded" />
                                    <div className="flex-1">
                                        <textarea value={m.styleDesc || ''} onChange={(e) => updateModelDesc(m.id, e.target.value)} className="w-full bg-slate-950 border border-slate-700 rounded p-2 text-xs text-white outline-none resize-none h-16" placeholder={`Pakaian Model ${index+1}...`}/>
                                    </div>
                                </div>
                            ))}
                        </div>
                    </div>
                ) : (
                    <div className="animate-in fade-in slide-in-from-bottom-2 duration-300">
                         <div className="bg-slate-950 p-4 rounded-lg border border-slate-800 mb-4">
                            <div className="grid grid-cols-2 gap-4 mb-4">
                                <div><label className="text-[10px] font-bold text-slate-500 block mb-1.5 uppercase">Gender</label><select className="w-full bg-slate-900 border border-slate-700 rounded-lg p-2.5 text-xs text-white" value={aiModelConfig.gender} onChange={e=>setAiModelConfig({...aiModelConfig, gender: e.target.value})}><option value="auto">Otomatis</option><option value="female">Wanita</option><option value="male">Pria</option></select></div>
                                <div><label className="text-[10px] font-bold text-slate-500 block mb-1.5 uppercase">Usia</label><select className="w-full bg-slate-900 border border-slate-700 rounded-lg p-2.5 text-xs text-white" value={aiModelConfig.age} onChange={e=>setAiModelConfig({...aiModelConfig, age: e.target.value})}><option>Anak-anak</option><option>Remaja</option><option>Dewasa</option><option>Orang Tua</option></select></div>
                            </div>
                            <label className="flex items-center gap-3 cursor-pointer p-3 bg-slate-900 rounded-lg border border-slate-700 hover:border-yellow-500 group"><div className={`w-4 h-4 rounded border flex items-center justify-center ${aiModelConfig.hijab ? 'bg-yellow-600 border-yellow-600' : 'border-slate-500'}`}>{aiModelConfig.hijab && <Check size={10} className="text-white"/>}</div><input type="checkbox" checked={aiModelConfig.hijab} onChange={e=>setAiModelConfig({...aiModelConfig, hijab: e.target.checked})} className="hidden"/><span className="text-xs text-slate-300">Hijab?</span></label>
                        </div>
                    </div>
                )}
            </StepPanel>

            <button onClick={handleGenerateButton} disabled={loading} className={`w-full py-4 rounded-xl font-bold text-sm uppercase tracking-widest shadow-lg flex items-center justify-center gap-2 ${loading ? 'bg-slate-800 text-slate-500 cursor-wait' : 'bg-yellow-600 hover:bg-yellow-500 text-white'}`}>{loading ? <RefreshCw className="animate-spin" size={16}/> : <Sparkles size={16}/>}{loading ? "PROSES..." : "GENERATE ASET"}</button>
        </div>

        <div className="md:col-span-7 space-y-8">
            {['ugc'].map((cat) => (
                <div key={cat} className="space-y-3">
                    <div className="flex items-center gap-2 border-b border-slate-800 pb-2">
                        {cat === 'ugc' && <User size={16} className="text-green-500"/>}
                        <h2 className="text-sm font-bold text-white uppercase tracking-wider">{cat} Photos</h2>
                    </div>
                    <div className="flex gap-4 overflow-x-auto pb-4 custom-scrollbar">
                        {resultsReady ? generatedImages[cat].map((item, idx) => (
                            <ResultCard key={item.id} item={item} onViewImage={setViewImageUrl} productInfo={productInfo} selectedRatio={ratio} onRegenerate={() => handleSingleRegenerate(cat, idx)} />
                        )) : (
                            Array(3).fill(0).map((_, i) => (
                                <div key={i} className="shrink-0 bg-slate-900/50 rounded-lg border border-dashed border-slate-800 flex items-center justify-center" style={{ width: ratio === '16:9' ? '460px' : ratio === '1:1' ? '340px' : '280px', aspectRatio: ratio.replace(':','/') }}><ImageIcon className="text-slate-800" size={24}/></div>
                            ))
                        )}
                    </div>
                </div>
            ))}

            <div className="bg-slate-900 border border-slate-800 rounded-xl p-6 mt-8 transition-all">
                <div className="flex justify-between items-center cursor-pointer group" onClick={() => setIsStudioOpen(!isStudioOpen)}>
                    <div className="flex items-center gap-3"><div className="w-10 h-10 bg-yellow-500/10 rounded-full flex items-center justify-center text-yellow-400"><Mic size={20}/></div><div><h2 className="text-white font-bold">Voice Over Studio</h2></div></div>
                    <ChevronDown className={`transform transition-transform ${isStudioOpen ? 'rotate-180' : 'rotate-0'}`} />
                </div>
                {isStudioOpen && (
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mt-6 animate-in slide-in-from-top-2">
                        <div className="space-y-3">
                            <textarea value={voState.script} onChange={e => setVoState({...voState, script: e.target.value, audioUrl: null})} className="w-full bg-slate-950 border border-slate-800 rounded-lg p-3 text-sm h-32 resize-none outline-none" placeholder="Tulis naskah..."/>
                            <div className="grid grid-cols-2 gap-2">
                                <select className="bg-slate-950 border border-slate-800 text-xs rounded p-2 text-white" value={voState.voice} onChange={e=>setVoState({...voState, voice: e.target.value})}>{ttsVoices.map(v => (<option key={v.id} value={v.id}>{v.name}</option>))}</select>
                                <button onClick={handleGenerateAudio} disabled={voState.loadingAudio} className="bg-yellow-600 text-white text-xs font-bold rounded flex items-center justify-center gap-2">{voState.loadingAudio ? <RefreshCw className="animate-spin" size={14}/> : <Volume2 size={14}/>} Audio</button>
                            </div>
                        </div>
                        <div className="flex flex-col justify-center items-center bg-slate-950 border border-slate-800 rounded-lg p-6">
                            {voState.audioUrl ? (
                                <div className="space-y-4 w-full">
                                    <button onClick={toggleAudio} className="w-16 h-16 rounded-full bg-yellow-500 mx-auto flex items-center justify-center text-white">{voState.isPlaying ? <X size={24}/> : <Play size={24} fill="white" />}</button>
                                    <a href={voState.audioUrl} download="voiceover.wav" className="block w-full py-2 bg-slate-800 text-white text-xs font-bold rounded text-center">Download WAV</a>
                                </div>
                            ) : <p className="text-xs text-slate-600">Audio ready here</p>}
                        </div>
                    </div>
                )}
            </div>
        </div>
      </main>

      <footer className="mt-12 border-t border-slate-800 py-6 text-center">
           <p className="text-xs text-slate-500 font-mono">GREEN SAFIR | 2026</p>
      </footer>

      {viewImageUrl && (
          <div className="fixed inset-0 z-[60] bg-black/90 flex items-center justify-center p-4" onClick={() => setViewImageUrl(null)}>
              <img src={viewImageUrl} className="max-h-[90vh] max-w-full rounded shadow-2xl" onClick={e => e.stopPropagation()}/>
          </div>
      )}

      {showRecModal && (
          <div className="fixed inset-0 z-[60] bg-black/80 backdrop-blur-sm flex items-center justify-center p-4">
              <div className="bg-gray-800 border border-yellow-500 w-full max-w-lg rounded-xl p-6 shadow-2xl">
                  <h3 className="text-xl font-bold text-yellow-400 mb-4">Rekomendasi Pengisian</h3>
                  <div className="flex justify-end gap-4 mt-8">
                      <button onClick={handleSkipRecommendation} className="bg-gray-600 text-white py-2 px-6 rounded-full">Abaikan</button>
                      <button onClick={handleApplyRecommendation} disabled={isGeneratingDesc} className="bg-yellow-600 text-white py-2 px-6 rounded-full">{isGeneratingDesc ? '...' : 'Generate Deskripsi'}</button>
                  </div>
              </div>
          </div>
      )}
    </div>
  );
};

export default App;
